Quick picks

Use Go Webview (webview_go) when:

You already have a Go web project (Gin, Fiber, Echo) running SSR and want a fast package with a 10–15 MB installer.

Use Wails when:

You want a polished desktop app with a new React/Vue/Vite UI and smooth Go system-level logic.

Use Lorca when:

You want to avoid WebView quirks on older Windows and lean on the user’s installed Chrome for consistent rendering.

Core idea:

The app runs a tiny web server on the user’s machine (loopback only at 127.0.0.1). WebView acts like a mirror without a browser address bar, so it feels like a native desktop app.

webview_go (minimal SSR) 
package main

import (
	"embed"
	"html/template"
	"net"
	"net/http"
	"github.com/webview/webview_go"
)

//go:embed templates/*
var templatesFS embed.FS

func main() {
	// 1. Pick a random free port
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		panic(err)
	}
	port := listener.Addr().(*net.TCPAddr).Port

	// 2. Run SSR server (Gin, Fiber, or std HTTP)
	tmpl := template.Must(template.ParseFS(templatesFS, "templates/*.html"))
	http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		tmpl.ExecuteTemplate(w, "index.html", map[string]interface{}{
			"Title": "Welcome to Go SSR Desktop!",
		})
	})

	go http.Serve(listener, nil)

	// 3. Open WebView pointed at the local server
	w := webview.New(false)
	defer w.Destroy()
	w.SetTitle("Go SSR Desktop App")
	w.SetSize(1024, 768, webview.HintNone)
	w.Navigate(http.Sprintf("http://127.0.0.1:%d", port))
	w.Run()
}
Wails v2/v3 (recommended) 
// Install Wails CLI:
// go install github.com/wailsapp/wails/v2/cmd/wails@latest
// Init: wails init -n my-app -t svelte

package main

import (
	"embed"
	"github.com/wailsapp/wails/v2"
	"github.com/wailsapp/wails/v2/pkg/options"
	"github.com/wailsapp/wails/v2/pkg/options/assetserver"
)

//go:embed all:frontend/dist
var assets embed.FS

func main() {
	app := NewApp() // App struct with your logic

	err := wails.Run(&options.App{
		Title:  "Wails Pro App",
		Width:  1024,
		Height: 768,
		AssetServer: &assetserver.Options{
			Assets: assets, // Wails serves assets; no hidden HTTP server needed
		},
		Bind: []interface{}{
			app, // Expose Go methods to JavaScript
		},
	})

	if err != nil {
		println("Error:", err.Error())
	}
}
